#Handles scandals, ends the event chain if things go to far
character_event = {
	id = synod.100
	hide_window = yes
	is_triggered_only = yes
	trigger = { OR = {has_character_flag = gathering_of_the_synod has_character_flag = gathering_of_the_ecu_council}}
	immediate = {
		if = {
			limit = {
				NOR = {
					has_character_flag = synod_minor_scandal
					has_character_flag = synod_major_scandal
				}
			}
			set_character_flag = synod_minor_scandal
			break = yes
		} 
		if = {
			limit = { has_character_flag = synod_minor_scandal }
			clr_character_flag = synod_minor_scandal
			set_character_flag = synod_major_scandal
			break = yes
		}

		if = {
			limit = { has_character_flag = synod_major_scandal }
			clr_character_flag = synod_major_scandal

			character_event = { id = synod.101 }
		}
	}
}
