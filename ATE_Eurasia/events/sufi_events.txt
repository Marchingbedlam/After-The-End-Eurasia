character_event = {
	id = sufi.2
	desc = EVTDESC_sufi_2
	picture = GFX_evt_religious_exultation

	is_triggered_only = yes
	
	option = {
		name = EVTOPTA_sufi_2
		set_character_flag = sufi_martial
		if = {
			limit = {
				is_sunni = yes
			}
			character_event = {id = sufi.3 }
		}
		else_if = {
			limit = {
				is_shiite = yes
			}
			character_event = {id = sufi.5 }
		}
	}
	option = {
		name = EVTOPTB_sufi_2
		set_character_flag = sufi_music_poetry
		trigger = {
			OR = {
				is_shiite = yes
				religion = hui
			}
		}
		if = {
			limit = {
				religion = hui
			}
			character_event = {id = sufi.4 }
		}
		else_if = {
			limit = {
				is_shiite = yes
			}
			character_event = {id = sufi.6 }
		}
		else = {
			limit = {
				AND = {
					is_sunni = yes
					NOT = {religion = hui}
				}
			}
			character_event = {id = sufi.9 }
		}
	}
	option = {
		name = EVTOPTC_sufi_2
		set_character_flag = sufi_mysticism
		if = {
			limit = {
				is_shiite = yes
			}
			character_event = {id = sufi.7 }
		}
		else_if = {
			limit = {
				is_sunni = yes
			}
			character_event = {id = sufi.8 }
		}
	}
}
