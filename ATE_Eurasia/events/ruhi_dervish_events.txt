# Start looking for dervish

	id = ruhi_dervish.3000
	desc = EVTDESC_ruhi_dervish_3000

	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

		name = EVTOPTA_ruhi_dervish_3000
		hidden_tooltip = { character_event = { id = ruhi_dervish.3001 days = 30 random = 60 } }
	}
}

# dervish is found
character_event = {
	id = ruhi_dervish.3001

	hide_window = yes
	
	is_triggered_only = yes

	immediate = {
		clr_character_flag = looking_for_dervish
		set_character_flag = found_dervish
		random_list = {
			15 = { # Finds great dervish
				create_character = {
					random_traits = no
					dynasty = NONE
					religion = ROOT
					culture = ROOT
					female = no
					age = 50
					attributes = {
						diplomacy = 12
						stewardship = 10
						intrigue = 11
						learning = 14
					}
					trait = genius
					trait = scholar
					trait = patient
					trait = mastermind_theologian
					trait = mystic
					trait = celibate
					trait = chaste
				}
				new_character = {
					set_character_flag = guru
					give_minor_title = title_dervish
					piety = 600
					prestige = 150
					opinion = {
						modifier = opinion_loyal_servant
						who = ROOT
						years = 100
					}
					random_list = {  # Small chance of having some unusual memorable trait
						35 = { add_trait = mevlevi }
						1 = { add_trait = blinded }
						1 = { add_trait = dwarf }
						50 = {}
					}
					character_event = { id = ruhi_dervish.3002 }
				}
			}
			80 = { # Finds ordinary dervish
				create_character = {
					random_traits = no
					dynasty = NONE
					religion = ROOT
					culture = ROOT
					female = no
					age = 50
					attributes = {
						diplomacy = 10
						stewardship = 8
						intrigue = 9
						learning = 12
					}
					trait = quick
					trait = scholar
					trait = patient
					trait = mastermind_theologian
					trait = mystic
					trait = chaste
				}
				new_character = {
					set_character_flag = guru
					give_minor_title = title_dervish
					piety = 300
					prestige = 750
					opinion = {
						modifier = opinion_loyal_servant
						who = ROOT
						years = 100
					}
					random_list = {  # Small chance of having some unusual memorable trait
						15 = { add_trait = mevlevi }
						25 = { add_trait = celibate }
					}
					character_event = { id = ruhi_dervish.3002 }
				}
			}
			5 = { # Finds con artist pretending to be dervish - but doesn't know it yet...
				create_character = {
					random_traits = no
					dynasty = NONE
					religion = ROOT
					culture = ROOT
					female = no
					age = 40
					attributes = {
						diplomacy = 12
						stewardship = 8
						intrigue = 16
						learning = 10
					}
					trait = quick
					trait = dutiful_cleric
					trait = weak
				}
				new_character = {
					set_character_flag = guru
					set_character_flag = fake_guru
					give_minor_title = title_dervish
					piety = 100
					prestige = 50
					add_trait = brahmin
					opinion = {
						modifier = opinion_loyal_servant
						who = ROOT
						years = 100
					}
					character_event = { id = ruhi_dervish.3002 }
				}
			}
		}
	}
}

# Drunkard
character_event = {
	id = ruhi_dervish.3100
	desc = EVTDESC_ruhi_dervish_3100
	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes
	
	ai = no
	only_capable = yes
	prisoner = no
	min_age = 16
	religion = ruhi


	trigger = {
		religion = ruhi
		has_dlc = "The Sword of Islam"
		any_courtier = {
			has_minor_title = title_dervish
			NOT = { has_character_flag = fake_guru }
		}
		trait = drunkard
	}
	
	option = {
		name = EVTOPTA_ruhi_dervish_3100
		prestige = -20
		random = {
			chance = 10
			remove_trait = drunkard
		}
	}
	option = {
		name = EVTOPTB_ruhi_dervish_3100
		piety = -20
		any_courtier = {
			limit = { has_minor_title = title_dervish }
			opinion = {
				modifier = opinion_unhappy
				who = ROOT
				years = 1
			}
		}
	}
}

	id = ruhi_dervish.3300

	hide_window = yes
	
	ai= no
	min_age = 16
	only_capable = yes
	prisoner = no
	religion = ruhi
	
	is_triggered_only = yes

	trigger = {
		has_dlc = "The Sword of Islam"
		any_courtier = {
			has_minor_title = title_dervish
			trait = genius
		}
		NOT = { is_inaccessible_trigger = yes }
	}
	
	immediate = {
		if = {
			limit = {
				any_courtier = {
					has_minor_title = title_dervish
				}
			}
			random_courtier = {
				limit = { has_minor_title = title_dervish }
				character_event = { id = ruhi_dervish.3301 }
			}
		}
		if = {
			limit = {
				any_courtier = {
					has_minor_title = title_dervish
					NOT = { trait = genius }
				}
			}
			random = {
				chance = 10
				random_courtier = {
					limit = { has_minor_title = title_dervish }
					character_event = { id = ruhi_dervish.3301 }
				}
			}
		}
	}
}
