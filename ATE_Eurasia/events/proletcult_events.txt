narrative_event = {
	id = proletcult.2
	title = EVTTITLE_proletcult2
	desc = EVTDESC_proletcult2
	picture = GFX_evt_victory_day
	
	is_triggered_only = yes
	
	hide_from = yes
	
	option = {
		name = EVTOPTA_proletcult2
		any_vassal = {
			limit = {
			    OR = { 
				    religion = proletcult
					culture = severyane
				    culture = muscovite
					culture = pomor
					culture = volgan
					culture = chernyan
					culture = uralic
					culture = siberian
					culture = akmolensk
					culture = kaliningradyets
				}
				prisoner = no
				war = no
				NOT = { trait = incapable }
			}
			character_event = { id = proletcult.3 days = 2 tooltip = EVTTOOLTIP_proletcult2 }
		}
		custom_tooltip = { text = victory_day_invitations_sent }
		hidden_tooltip = { character_event = { id = proletcult.5 days = 5 } }
		hidden_tooltip = {
		add_character_modifier = {
			name = held_victory_day_timer
			duration = 360
			hidden = yes
		}
		}
	}
}

# Invitation Received
character_event = {
	id = proletcult.3
	desc = EVTDESC_proletcult3
	picture = GFX_evt_victory_day
	border = GFX_event_normal_frame_religion
	
	is_triggered_only = yes

	trigger = { NOT = { is_inaccessible_trigger = yes } }
	
	option = {
		name = EVTOPTA_proletcult3
		ai_chance = {
			factor = 95
			modifier = {
				factor = 0
				opinion = { who = ROOT value <= -20 }
			}
		}
		custom_tooltip = { text = victory_day_accepted }
		set_character_flag = attending_victory_day
		set_character_flag = do_not_disturb
		hidden_tooltip = {
		    add_character_modifier = {
			   name = held_victory_day_timer
			   duration = 360
			   hidden = yes
		    }
		}
		hidden_tooltip = { character_event = { id = proletcult.5 days = 3 } }
	}
	option = {
		name = EVTOPTB_proletcult3
		ai_chance = {
			factor = 5
		}
		FROM = {
			opinion = {
				modifier = opinion_refused_victory_day
				who = ROOT
			}
		}
		custom_tooltip = { text = victory_day_declined }
		hidden_tooltip = {
			FROM = { character_event = { id = proletcult.4 } }
		}
	}
}
