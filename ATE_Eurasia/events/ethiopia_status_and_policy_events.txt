# New Ethiopian Emperor. Pick a new Policy and send a News Event!
# Hidden. Fired from on_offmap_ruler_changed.
character_event = {
	id = ED.50000
	offmap = only

	has_dlc = "Jade Dragon"

	is_triggered_only = yes
	hide_window = yes

	trigger = { is_offmap_tag = offmap_ethiopia }
    
	immediate = {
		FROMFROM = {
			set_offmap_flag = no_policy_news # Don't fire the News event in 'on_offmap_policy_changed'
			random_list = {
				100 = { # OPEN
					modifier = {
						factor = 1.5
						has_status = ethiopia_stable
					}
					modifier = {
						factor = 5
						has_policy = ethiopia_open
						NOT = { days_since_policy_change = 10950 } #30 years
					}
					set_policy = ethiopia_open
					log = "-------------------------------------"
					log = "Ethiopia Policy Logging:"
					log = "Ethiopias new Policy is now OPEN"
					log = "-------------------------------------"
				}
				20 = { # ISOLATIONIST
					trigger = {
						OR = {
							NOT = { has_offmap_tmp_flag = ethiopia_forced_open }
							had_offmap_tmp_flag = { flag = ethiopia_forced_open years = 50 }
						}
					}
					modifier = {
						factor = 10
						has_offmap_flag = ethiopia_isolationist_trend
					}
					modifier = { # Hard times... (negative Status)
						factor = 2
						ethiopia_is_suffering_trigger = yes
					}
					modifier = {
						factor = 0.5
						has_status = ethiopia_golden_age
					}
					modifier = {
						factor = 5
						has_policy = ethiopia_isolationist
						NOT = { days_since_policy_change = 10950 } #30 years
					}
					set_policy = ethiopia_isolationist
					log = "-------------------------------------"
					log = "Ethiopia Policy Logging:"
					log = "Ethiopias new Policy is now ISOLATIONIST"
					log = "-------------------------------------"
					clr_offmap_flag = ethiopia_isolationist_trend
					clr_offmap_tmp_flag = ethiopia_forced_open
				}
				10 = { # EXPANSIONIST
					trigger = {
						NOT = { has_offmap_flag = ethiopia_badly_beaten_3 }
						NOT = { check_variable = { which = "global_ethiopia_change_to_expansionist_count" value = 6 } }
					}
					modifier = {
						factor = 20
						has_offmap_flag = ethiopia_expansionist_trend
					}
					modifier = {
						factor = 2
						has_status = ethiopia_golden_age
					}
					modifier = {
						factor = 0.75
						has_offmap_flag = ethiopia_badly_beaten_1
					}
					modifier = {
						factor = 0.5
						has_offmap_flag = ethiopia_badly_beaten_2
					}
					modifier = {
						factor = 10
						has_policy = ethiopia_expansionist
						NOT = { days_since_policy_change = 14600 } #40 years
					}
					modifier = {
						factor = 0.1
						has_policy = ethiopia_expansionist
						days_since_policy_change = 32850 #90 years
					}
					modifier = {
						factor = 0.2
						check_variable = { which = "global_ethiopia_change_to_expansionist_count" value = 3 }
					}
					if = {
						limit = { NOT = { has_policy = ethiopia_expansionist } }
						change_variable = { which = "global_ethiopia_change_to_expansionist_count" value = 1 }
					}
					set_policy = ethiopia_expansionist
					log = "-------------------------------------"
					log = "Ethiopia Policy Logging:"
					log = "Ethiopias new Policy is now EXPANSIONIST"
					log = "-------------------------------------"
					clr_offmap_flag = ethiopia_expansionist_trend
				}
			}
		}
		
		if = {
			limit = { FROMFROM = { NOT = { has_offmap_flag = no_succession_news } } }
			any_player = {
				limit = {
					has_offmap_news_enabled = FROMFROM
					is_within_diplo_range = ROOT
				}
				narrative_event = { id = ED.50001 }
			}
		}
		
		FROMFROM = {
			clr_offmap_flag = no_policy_news
			clr_offmap_flag = no_succession_news
		}
	}
}

				}
			}
		}
		modifier = { # Slightly more likely to get Unrest if non-Habeshan emperor
			factor = 1.25
			FROM = {
				NOT = {
					culture_group = habesha
				}
			}
