	is_triggered_only = yes
	
	option = {
		name = EVTOPTA_pravoslavie.8
		if = {
		   random_list = {
		     10 = {
			    ROOT = { change_martial = 1 }
			 }
			 5 = {
			    ROOT = { change_martial = 2 }
			 }
			 85 = { } 
		   }
		}
	}
	
	option = {
		name = EVTOPTB_pravoslavie.8
		trigger = { NOT = { trait = celibate trait = eunuch } }
		add_character_modifier = {
			name = pravoslavie_ask_fertility
			years = 10
		}
	}
	
	option = { 
		name = EVTOPTC_pravoslavie.8
		if = {
		   random_list = {
		     15 = {
			    ROOT = { change_stewardship = 1 }
			 }
			 10 = {
			    ROOT = { change_stewardship = 2 }
			 }
			 75 = { } 
		   }
		}
	}
	
	option = {
		name = EVTOPTD_pravoslavie.8
		piety = 50
		if = {
		   limit = { NOT = { trait = humble } }
		   random_list = {
		     25 = {
			    add_trait="humble"
			 }
			 75 = { } 
		   }
		}
	}
}
