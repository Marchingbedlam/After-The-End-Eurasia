namespace = LoR

# Augustus Trait inheritance
character_event = {
	id = LoR.47

	is_triggered_only = yes # on_death
	hide_window = yes

	only_independent = yes
	has_global_flag = rome_restored
	has_dlc = "Legacy of Rome"

	trigger = {
		has_landed_title = e_roman_empire
	}

	immediate = {
		e_roman_empire = {
			current_heir = {
				character_event = {
					id = LoR.45
					days = 2
				}
			}
		}
	}
}

		}
