		}
	}
	desc = {
		text = EVTDESC_COSSACK_7_CONSTANTINOPOLE
		picture = GFX_evt_constantinopole
		trigger = { 
			has_character_flag = go_to_constantinopole_kozachina
		}
	}
	desc = {
		text = EVTDESC_COSSACK_7_KYIV
		picture = GFX_evt_kyiv
		trigger = {
			has_character_flag = go_to_kyiv_kozachina
		}
	}

	is_triggered_only = yes
	
	option = { 
		name = EVTOPTA_COSSACK_7
		hidden_tooltip = { 
		    character_event = { id = SoA.5055 days = 7 }
		    if = {
		         limit = { has_character_flag = go_to_constantinopole_kozachina NOT = { trait = "kozachina_st_george } }
		         character_event = { id = cossack.8 days = 1 }
		    }
		    if = {
		         limit = { has_character_flag = go_to_zaporizhzhia_kozachina }
		         character_event = { id = cossack.10 days = 1 }
		    }
		    if = {
		         limit = { has_character_flag = go_to_kyiv_kozachina }
		         character_event = { id = cossack.9 days = 1 }
		    }
			clr_character_flag = go_to_constantinopole_kozachina
		    clr_character_flag= go_to_zaporizhzhia_kozachina
		    clr_character_flag = go_to_kyiv_kozachina
		}
	}
}
